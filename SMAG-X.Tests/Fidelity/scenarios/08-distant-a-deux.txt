# Le dépôt distant à deux : clone d'un projet vide, push, suivi des branches, push refusé, fetch, pull divergent, branche partagée
forge classe/projet
$ git config --global user.name "Alice Martin"
$ git config --global user.email alice@lycee.example
$ cd ..
$ git clone git@gitlab.example:classe/projet.git
$ cd projet
$ git status
$ git remote
$ git remote -v
write index.html <h1>Accueil</h1>\n
$ git add index.html
$ git commit -m "Accueil"
$ git status
$ git branch -vv
$ git push
$ git status
$ git branch -vv
$ git push -u origin main
append index.html <p>Bienvenue</p>\n
$ git commit -am "Bienvenue"
$ git status
$ git branch -v
$ git branch -vv
$ git push
$ git push

$ git config --global user.name "Bob Durand"
