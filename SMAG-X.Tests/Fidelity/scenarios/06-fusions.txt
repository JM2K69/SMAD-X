# Fusions : plusieurs fichiers, ajouts des deux côtés, conflit add/add, modify/delete, --abort, --no-ff, « into »
$ git config --global user.name "Alice Martin"
$ git config --global user.email alice@lycee.example
$ git config --global init.defaultBranch main
$ git init
write index.html <html>\n<head>\n<title>Accueil</title>\n</head>\n<body>\n<h1>Bienvenue</h1>\n</body>\n</html>\n
write style.css body {\n  margin: 0;\n}\n
write contact.html <p>Contact</p>\n
$ git add .
$ git commit -m "Site"
$ git switch -c design
write style.css body {\n  margin: 0;\n  color: navy;\n}\nh1 {\n  font-size: 2em;\n}\n
write logo.svg <svg/>\n
$ git add .
$ git commit -m "Couleurs et logo"
$ git switch main
write index.html <html>\n<head>\n<title>Accueil</title>\n<meta charset="utf-8">\n</head>\n<body>\n<h1>Bienvenue</h1>\n</body>\n</html>\n
$ git commit -am "Encodage"
$ git merge design
$ git log --oneline
$ git switch -c pied
